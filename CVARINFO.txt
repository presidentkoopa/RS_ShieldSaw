// RS_ShieldSaw, standalone again (2026-09-21). It lived inside RS_VR_Weapons from 09-14 to 09-21;
// the owner took it back out so it plays with any weapon pack on its own. Every name is unchanged,
// so saved values and bindings carry over from the folded build.


// THE FOLD (09-14, the build lane's ruling, RS_Grenade ffff0c6's twin): the game rules below are SERVER --
// rs_ss_start, rs_ss_cut_damage, rs_ss_throw_speed, rs_ss_throw_min, rs_ss_locks, rs_ss_lock_cone, rs_ss_lock_range, rs_ss_deflect, rs_ss_deflect_stowed.
// Everything else stays `user`, which CVARINFO makes a userinfo cvar (d_main.cpp ParseCVarInfo), so the
// per-player choices this code reads through the acting player (equip, gesture, mount_*) already travel
// to every machine; placement and debug are presentation.

// RS_ShieldSaw.

// Start the game with it, no pickup needed.
server bool  rs_ss_start       = true;

// Drive the sequence from the off-hand GRIP: squeeze to draw it off the
// forearm, hold to keep it, release to throw. Off falls back to the bound key.
user bool  rs_ss_gesture     = true;

// Start it already DRAWN rather than stowed. Off by default: stowed is the
// resting state and the one the draw/grind/throw sequence begins from.
user bool  rs_ss_equip       = false;

// Hand model on the held shield.
user bool  rs_ss_handmodel   = true;

// ---- lock-on -------------------------------------------------------------
server int   rs_ss_locks       = 5;        // targets one throw can visit
server float rs_ss_lock_cone   = 14.0;     // acquire half-angle, from the HAND
server float rs_ss_lock_range  = 1200.0;

// ---- throw ---------------------------------------------------------------
server float rs_ss_throw_speed = 1.0;
server float rs_ss_cut_damage  = 1.0;

// HOW FAST YOUR HAND HAS TO BE MOVING, in map units per second (NOT per tic --
// see HandMoving() in rs_shieldsaw_state.zs), for a release away from the
// mount to count as a throw instead of just putting it down. Below this, a
// release reads as "I let go without meaning to send it anywhere" and the
// shield goes back to the mount instead of flying off a slow, careless
// release. In forearm mode this is the ONLY thing that decides throw vs.
// stow, since the forearm has no separate "put it back" spot to release at.
// vr_vunits_per_meter is ~34 by default, so 25 is a shade under 3/4 m/s.
server float rs_ss_throw_min   = 25.0;

// ---- passive -------------------------------------------------------------
// The deflector exists only while the shield is IN YOUR HAND. Throwing it
// costs you the guard; that is the point.
server bool  rs_ss_deflect     = true;

// Whether the shield still deflects while STOWED -- on the off forearm or the
// shoulder, whichever rs_ss_mount_mode has it on. On is the point of a mount
// you can leave a guard sitting on; off means only a shield actually in your
// hand blocks anything.
server bool  rs_ss_deflect_stowed = true;

// HOW BIG THE GUARD IS, and how far off the hand it sits.
//
// The shield in your hand is 19.7 map units across, so its own rim is a radius of about 10. 14 is
// that plus the slack a hand that is moving needs for raising the thing to be worth doing. The
// standoff pushes the face off the palm so the disc guards in FRONT of your fist rather than
// through it.
//
// These replaced an invisible +SHOOTABLE box. Nothing shootable exists any more, so making this
// bigger can no longer eat your own fire -- it only changes what you can block.
server float rs_ss_guard_radius   = 14.0;
server float rs_ss_guard_standoff = 6.0;

// WHERE A TURNED SHOT GOES. On, back at whoever fired it, which is the whole fantasy. Off, mirrored
// off the shield's face like a ball off a wall -- which is the honest physics and much harder to
// aim with.
server bool  rs_ss_deflect_aim    = true;

// ---- the bash (CODER_PLAN step 54) ---------------------------------------
//
// SHOVE THE SHIELD INTO SOMETHING AND IT HITS. The shield had no offence at all: the guard
// turns missiles, the disc cuts when thrown, and driving it into a face did nothing -- which
// is the first thing anybody tries with a shield. Owner: "wants its behaviour MEANER".
//
// It also fills the hole that made the shield look like it only stops bullets. The guard
// iterates the engine's MISSILE list, and a monster's claw is not a missile, so nothing in
// this package had ever looked at a monster.
//
// _speed is measured ALONG THE SHIELD'S FACE, in map units a second, so swinging it sideways
// or dropping your arm is not a bash however fast it is. 70 is a deliberate shove rather than
// a walk. Raise it if you bash by accident, lower it if a real shove is ignored.
//
// Damage is clamp(10 + 0.15 x speed, 10, 40) x _damage, with a 10-tic cooldown PER TARGET so
// one push cannot hit the same monster six tics running while still letting you catch the one
// beside it. SERVER cvars, not user: they decide damage.
server bool  rs_ss_bash        = true;
server float rs_ss_bash_speed  = 70.0;
server float rs_ss_bash_damage = 1.0;

// ---- what a thrown disc does, by what it hit (CODER_PLAN step 55) --------
//
// The disc used to do one thing to everything: roll 24-44 and rip onward, so a zombie and a
// baron felt identical and the throw had no read to it. Now:
//
//   fodder  (SpawnHealth <= _tier_fodder)  BISECT -- comes apart. +NOEXTREMEDEATH is dropped
//                                          for that hit only and the damage is forced past
//                                          the gib threshold, so it happens every time
//                                          rather than on a lucky roll. (Owner, Q43: "a
//                                          thrown saw blade that leaves a body intact reads
//                                          wrong.")
//   mid     (<= _tier_mid)                 EMBED -- the disc buries itself, rides the body
//                                          and grinds _embed_dps a tic for _embed_tics,
//                                          then flies home.
//   heavy   (above that, or +BOSS)         GLANCE -- bounces off at _glance_scale damage. A
//                                          shield saw does not bury itself in a baron.
//
// CLASSIFIED ON SpawnHealth, NEVER ON CLASS NAMES: a modded imp is still fodder and a mod's
// own boss is still a boss, and a name list would be wrong the first time a monster pack is
// loaded. SpawnHealth is the AUTHORED health, so a baron on its last hit point is still a
// baron and does not suddenly bisect.
//
// 70 and 600 are the plan's guesses at where Doom's roster divides. Only play can say.
// _tiers off restores exactly the old single behaviour.
server bool  rs_ss_tiers        = true;
server int   rs_ss_tier_fodder  = 70;
server int   rs_ss_tier_mid     = 600;
server int   rs_ss_embed_tics   = 21;
server int   rs_ss_embed_dps    = 3;
server float rs_ss_glance_scale = 0.4;

// ---- the shoulder mount --------------------------------------------------
// WHERE ON YOUR BACK the shield sits, and how close your off hand has to get
// to take it. fwd is negative because the headset sits at your EYES, so your
// own back is behind it; side is negative for the left shoulder; frac is a
// fraction of eye height, and 0.86 is about shoulder.
user float rs_ss_mount_fwd   = -7.0;
user float rs_ss_mount_side  = -8.0;
user float rs_ss_mount_frac  = 0.86;
user float rs_ss_mount_reach = 9.0;

// THE ANCHOR IS THE SHIELD'S TOP EDGE, NOT ITS CENTRE.
//
// Anchoring a disc this size by its middle parks it centred on your shoulder
// blade -- a giant plate behind your head. Anchoring by the top instead lets
// it hang down your back the way a slung shield actually does, while the point
// you REACH for stays at a normal over-the-shoulder height.
//
// This is how far the disc's centre sits below that grab point. Roughly the
// shield's own radius.
user float rs_ss_mount_drop  = 11.0;

user float rs_ss_mount_yaw   = 0.0;
user float rs_ss_mount_pitch = 0.0;
user float rs_ss_mount_roll  = 0.0;
user float rs_ss_mount_scale = 1.0;

// The wireframe diamond at the mount. 0 never, 1 while it matters (on your
// back, or in your hand waiting to go back), 2 always. It is sized to the
// reach radius, so the diamond IS the volume you have to put your hand in.
user int   rs_ss_marker      = 1;

// ---- mount: shoulder or forearm -------------------------------------------
// 0 shoulder (default): a world anchor over your off shoulder, reached by
//   placing your off hand there -- see the "Shoulder mount" section below.
// 1 forearm: the shield rides your own off hand directly, always drawn,
//   always in reach. Brought back now that offHandFree() (below, and in
//   rs_shieldsaw_state.zs) can tell an empty-hand squeeze from RS_Hands
//   already holding something -- that distinction is what the original
//   forearm mount shipped without, and why it used to fight every pickup.
// The one cost that was never a logic bug: strapped to your own forearm, the
// shield still covers the arm you're looking at. Shoulder does not have that
// problem, which is why it stays the default.
user int   rs_ss_mount_mode  = 0;

// ---- the stowed model ------------------------------------------------------
// Show the shield's stowed model at all -- on the off forearm or the
// shoulder, whichever rs_ss_mount_mode has active -- while it is not in your
// hand. Off makes it vanish into inventory between draws, in either mode.
// rs_ss_deflect_stowed (below) governs whether it still deflects while gone
// from view; the two are independent.
user bool  rs_ss_stow        = true;

// Keep it drawn even while held, so it can be positioned without putting it
// away first. Turn off when you are done.
user bool  rs_ss_stow_place  = false;

// WHERE ON THE FOREARM, and where in the hand. These are read LIVE by the
// renderer every frame through MODELDEF's PlacementCVars -- the engine looks
// the names up by prefix, so the prefix in MODELDEF and the names here have to

// block is what actually reads the rs_ss_stow_* prefix now; before
// rs_ss_mount_mode existed nothing did.
//
// A number found on a slider can be folded straight into MODELDEF's Offset and
// the slider zeroed with nothing moving; they are summed in the same translate.

// Stowed on the off forearm.
user float rs_ss_stow_ofs_x  = 0.0;
user float rs_ss_stow_ofs_y  = 0.0;
user float rs_ss_stow_ofs_z  = 0.0;
user float rs_ss_stow_yaw    = 0.0;
user float rs_ss_stow_pitch  = 0.0;
user float rs_ss_stow_roll   = 0.0;
user float rs_ss_stow_scale  = 1.0;

// Held in the hand.
user float rs_ss_held_ofs_x  = 0.0;
user float rs_ss_held_ofs_y  = 0.0;
user float rs_ss_held_ofs_z  = 0.0;
user float rs_ss_held_yaw    = 0.0;
user float rs_ss_held_pitch  = 0.0;
user float rs_ss_held_roll   = 0.0;
user float rs_ss_held_scale  = 1.0;

// Print the gesture's decision on every grip edge, plus the raw button once a
// second. Tells apart "the grip never registered" from "the arbiter refused".
user bool  rs_ss_debug       = false;

// ---- A THING IN THE ROOM, NOT ON THE SCREEN -------------------------------
//
// Draw the shield as a world actor on the controller rather than as a player
// sprite. A psprite cannot be occluded by a doorframe, cannot take room light,
// and cannot be seen by another player -- which for a shield is most of the
// point of holding one up.
//
// Off puts it back on the psprite, which is the fallback if a controller frame
// is unavailable.
user bool rs_ss_world = true;

// ---- WHERE THE WORLD PROP SITS ON THE CONTROLLER --------------------------
//
// PlacementCVars reads these live, which is what lets the shield be dialled in
// while wearing the headset -- the renderer picks them up every frame, so the
// model moves as the slider does. NOTHING IN ZSCRIPT MAY WRITE THEM: two
// writers on one transform reads exactly like a dead slider.
//
// Two sets because the two hands hold it at different angles and sharing one
// prefix would mean tuning one and untuning the other.
user float rs_ss_prop_ofs_x = 0.0;
user float rs_ss_prop_ofs_y = 0.0;
user float rs_ss_prop_ofs_z = 0.0;
user float rs_ss_prop_yaw   = 0.0;
user float rs_ss_prop_pitch = 0.0;
user float rs_ss_prop_roll  = 0.0;
user float rs_ss_prop_scale = 1.0;

user float rs_ss_prop_off_ofs_x = 0.0;
user float rs_ss_prop_off_ofs_y = 0.0;
user float rs_ss_prop_off_ofs_z = 0.0;
user float rs_ss_prop_off_yaw   = 0.0;
user float rs_ss_prop_off_pitch = 0.0;
user float rs_ss_prop_off_roll  = 0.0;
user float rs_ss_prop_off_scale = 1.0;

// ---- THE THROWN SAW SPINS ---------------------------------------------------
//
// Taken off your wrist at the moment of release, through RS_WorldHands' shared
// throw measurement. Before this the shield left your hand at a fixed angle and
// held it for the whole flight -- a facing, not a spin, and a disc that does not
// turn is the gesture failing to land.
//
// Needs RS_WorldHands loaded; without it there is simply no spin and the shield
// flies as it always did.
user bool  rs_ss_spin = true;
user float rs_ss_spin_scale = 1.0;

// AND A FLOOR UNDER IT, in degrees per tic. A wrist roll of zero is the common case -- no
// RS_WorldHands, or simply a straight overhand throw with no twist in it -- and taking that
// literally left the disc facing one way for its whole flight. The wrist rides on top of this
// rather than replacing it.
//
// SERVER as of 2026-09-28, and 31 rather than 26. It stopped being a
// preference the moment the glide started reading it: the spin now holds the
// disc up, so this decides WHERE A THROWN SHIELD LANDS and every machine has
// to agree on it.
//
// 31 IS MEASURED, NOT PICKED. The teeth are painted into the alpha of
// shieldsaw_HD.png -- the mesh carries no usable periodicity at all -- and
// there are SIXTEEN of them, agreed by both texture rings that cross them at
// full contrast. So the disc looks IDENTICAL every 22.5 degrees, and any
// per-frame rotation near a multiple of that wagon-wheels: it reads as slow,
// stopped or backwards while spinning perfectly correctly.
//
// 66.1 IS THE SPECTACLE, CHOSEN KNOWINGLY (owner, 2026-09-28: "set spin to 66,
// i want the spectacle"). 6.43 revolutions a second, and the fastest rate in
// that band that is not near a tooth step:
//
//    72 Hz -> 32.13 deg/frame, 9.63 clear
//    90 Hz -> 25.71 deg/frame, 3.21 clear
//   120 Hz -> 19.28 deg/frame, 3.22 clear
//   144 Hz -> 16.07 deg/frame, 6.43 clear
//
// The margin is 3.21 of the 11.25 available, against 7.5 for the safest rate
// (31). That is the trade: it spins twice as fast and sits closer to a strobe
// at 90 and 120 Hz. If the disc ever reads as crawling or turning backwards,
// this number is why -- 31 is the safe one, and 37.1 is the middle.
//
// NOT 60, which was the original proposal and is the worst of all of them:
// 23.33 degrees a frame at 90Hz is 0.83 from a whole tooth step, so six
// revolutions a second would have LOOKED like a disc creeping forward, on the
// commonest headset rate there is.
//
// Run CardPipeline/tools/disc_teeth_texture.py before changing this.
server float rs_ss_spin_base = 66.1;

// HOW LONG A SQUEEZE KEEPS ASKING FOR THE SHIELD, in tics.
//
// The draw used to be a single-tic edge: it fired only if your hand was inside the mount's reach
// AND the grip arbiter answered "free" on the exact tic your finger closed, and a press that missed
// either was spent -- you squeezed, nothing happened, and the only way to retry was to let go. The
// window keeps asking while you hold the grip. 10 tics is under a third of a second.
user float rs_ss_press_window = 10.0;

// ---- THROWN LIKE A DISC (added 2026-09-28) --------------------------------

// What the disc weighs, for the launch speed. It keeps
// armKg / (armKg + this) of the hand's speed, so at 5 kg off a 30 kg arm the
// throw you make is 86% of the throw you get.
server float rs_ss_mass_kg = 5.0;

// The slowest a free throw can leave at, units per tic. The old code clamped
// into a band around Speed, which flattened a lob and a hurl into each other;
// this is the only part of that clamp worth keeping. A shield that cannot
// reach anything is not a weapon.
server float rs_ss_speed_floor = 6.0;

// How fast a free-flying disc sinks, units per tic squared. It is NOGRAVITY,
// so nothing else pulls on it at all. 0.11 is about 40% of real gravity at
// hand scale -- a disc falls, but it takes its time about it.
server float rs_ss_fall = 0.11;

// How much the spin holds it up, against that fall. Scaled by how hard it is
// spinning, how fast it is still going forward, and how FLAT the throw was --
// so a sidearm throw carries and one made with the disc on edge drops. Capped
// below rs_ss_fall, so lift can flatten a descent and never become a climb.
server float rs_ss_lift = 0.09;

// [SHIELDFACE] The shield face follows the hand ROLL, so holding it flat in front of you
// guards -- not only when the wrist is cocked as if mid-swat. Off restores the old
// yaw+pitch-only face, for comparing the two in one session.
user bool rs_ss_face_roll = true;

// Which way positive roll turns the face. Cannot be read off a file; flip it if the shield
// guards behind you instead of in front.
user bool rs_ss_face_invert = false;

// [GUARDDIAG] Say why the shield did or did not catch something: a haptic tick when a round
// is close enough to be considered, a strong pulse and a green flash on a catch, and a
// midprint naming the test that rejected it. For finding a fault in one pass instead of forty.
user bool rs_ss_guard_diag = true;

// [GUARDRING] DRAW THE HITBOX YOU ACTUALLY BLOCK WITH -- a red ring on the rim of the disc
// that was registered with the engine this tic, and a short green stub along the way its face
// looks.
//
// Four passes of shield work argued about where this disc was without once looking at it. The
// ring is not a decoration and not a guess: it is the registered volume itself, traced from
// the same three numbers the damage test is given. Where it sits beside the drawn shield IS
// the measurement, and one look settles the centre, the size and which way roll turns the
// face -- three things that between them cost days of inference.
//
// A yellow mark lands wherever the engine last recorded a block, so a parry that reports its
// point out in the open air can be seen doing it rather than argued about.
//
// On while the shield is being fixed. Turn it off when it is boring.
user bool rs_ss_guard_draw = true;

// [GUARDSEAT] WHERE THE BLOCKING DISC COMES FROM.
//
// On: the replicated usercmd latch, and the plate seat measured off the model -- 4.23 units
// to the hand left, 3.19 back toward the elbow, rim 9.90 (12.04 with the blades out).
// Off: the old behaviour exactly -- the renderer-written hand pose, a 6-unit push along the
// face normal and the rs_ss_guard_radius slider.
//
// A switch rather than a replacement so the two can be compared in one session. The old path
// described one hand with two writers sampled at different moments (position and aim from the
// renderer, roll from the latch), so while the arm moved the face and the centre disagreed.
user bool rs_ss_guard_latch = true;

// On: the rim is the mesh rim. Off: the rim is the rs_ss_guard_radius slider. Only has an
// effect while rs_ss_guard_latch is on.
user bool rs_ss_guard_mesh = true;

// [LEFTHAND] The shield mesh was authored for the LEFT hand and NOAUTOREVERSE stops the engine
// mirroring it, so in the right hand its front plate faces your own body. This mirrors the
// guard side to match. Set it if you play left-handed.
//
// A user cvar and not vr_control_scheme: that one is global config, so the playsim cannot read
// it for any player, including you.
user bool rs_ss_lefthanded = false;

// [EMPTYHAND] The owner's ruling, 2026-10-17: while the disc is in flight the off hand stays
// EMPTY. It is what makes a catch, a recall and a yank-back possible at all -- the disc used
// to fly home to a hand that was already holding your previous weapon, so the only thing it
// could do on arrival was disappear and turn up on your back.
//
// Off restores the old behaviour: your weapon comes straight back on the release.
user bool rs_ss_hand_empty = true;

// [CHARGE] Shove the shield forward while the plate faces the way you are already running and
// you dash behind it for a third of a second. The guard stays live through the dash, so a
// frontal attack during a charge is blocked by the thing you are charging with.
user bool rs_ss_charge = true;

// [BLOCKCHARGE] What you block spins the saw up, so taking a hit on the plate arms the blade.
user bool rs_ss_block_charge = true;

// [DENT] Blocked blows leave scorch marks on the plate, riding the shield's own model. Eight
// at a time, reused oldest first. Only blows above a light hit mark it.
user bool rs_ss_dents = true;
