// RS_ShieldSaw models.
//
// Meshes are the originals. shield.md3 is the shield body, frames 0 (closed)
// to 3 (fully open). shieldsaw.md3 is the blade: 0 stowed, 1-4 deploying,
// 5-7 spinning. hand.md3 is the hand, frame 1.
//
// SSAW carries the hand, SSNH does not. A_ShieldModel picks between them by
// writing the psprite's sprite index, so both variants share one state table
// instead of duplicating it. No graphics exist behind either name -- they are
// MODELDEF keys only.

// ---- held, with hand -----------------------------------------------------
Model RS_ShieldSaw
{
	Path "models/shieldsaw"
	Model 0 "shield.md3"
	Skin 0 "shieldsaw_b_HD.png"
	Model 1 "shieldsaw.md3"
	Skin 1 "shieldsaw_HD.png"

	Path "models/hand"
	Model 2 "hand.md3"
	Skin 2 "hand_HD.png"

	// NOAUTOREVERSE + a positive scale, matching the stow block that draws the
	// same mesh. The old `Scale -1` relied on the engine mirroring the model
	// for the non-dominant hand to cancel it out -- and that mirror is applied
	// only when the control scheme is right-handed, so a left-handed player got
	// the shield rendered inside out.
	Scale 1.0 1.0 1.0
	Offset 0.0 -24.0 -10.0
	NOAUTOREVERSE
	PlacementCVars rs_ss_held

	FrameIndex SSAW A 0 0	FrameIndex SSAW A 1 0	FrameIndex SSAW A 2 1
	FrameIndex SSAW B 0 1	FrameIndex SSAW B 1 1	FrameIndex SSAW B 2 1
	FrameIndex SSAW C 0 2	FrameIndex SSAW C 1 2	FrameIndex SSAW C 2 1
	FrameIndex SSAW D 0 3	FrameIndex SSAW D 1 3	FrameIndex SSAW D 2 1
	FrameIndex SSAW E 0 3	FrameIndex SSAW E 1 4	FrameIndex SSAW E 2 1
	FrameIndex SSAW F 0 3	FrameIndex SSAW F 1 5	FrameIndex SSAW F 2 1
	FrameIndex SSAW G 0 3	FrameIndex SSAW G 1 6	FrameIndex SSAW G 2 1
	FrameIndex SSAW H 0 3	FrameIndex SSAW H 1 7	FrameIndex SSAW H 2 1
}

// ---- in flight -----------------------------------------------------------
// SIXTEEN FRAMES, and they were all being thrown away. shield_weapon.md3
// carries a full spin baked into frames 0-15; the old block mapped every
// sprite letter to frame 0, so the shield slid through the air without
// turning at all.
//
// UseActorRoll is what makes it a Captain America throw rather than a fixed
// vertical one. The original bucketed the throw plane into seven states
// (RollOffset 0, +/-30, +/-60, +/-90) chosen from the wrist at release; taking
// the actor's own roll instead gives ANY angle, set once at launch from the
// hand and held for the whole flight.
//
// PitchFromMomentum is deliberately gone with it: it overwrote the pitch every
// frame from the velocity, which fights a throw plane the player chose.
//
// A SKIN PER SURFACE (fixed 2026-09-14). shield_weapon.md3's face and teeth
// (surfaces 0 and 2) are laid out on shieldsaw.png and only its back (surface 1)
// on shieldsaw_b.png, but a lone `Skin 0` dresses every surface -- the thrown
// shield flew with its face and teeth wearing the back plate's texture.
//
// THE SIZE OF THE SHIELD IN YOUR HAND (fixed 2026-09-14). Held, the shield is
// 57.94 mesh units across on the hand path, 57.94 x 0.34 = 19.70 map units.
// Thrown, this mesh is 56.56 across on the world path, and the actor's own
// Scale 0.55 made it 31.1 -- 1.6 times the shield that left your hand. 19.70 /
// 56.56 / 0.55 = 0.6333 brings it back to 19.70; the actor's 0.55 is left alone
// (RS_ShieldTrail shares it).
Model RS_ShieldInFlight
{
	Path "models/shieldsaw"
	Model 0 "shield_weapon.md3"
	Skin 0 "shieldsaw_HD.png"
	SurfaceSkin 0 0 "shieldsaw_HD.png"
	SurfaceSkin 0 1 "shieldsaw_b_HD.png"
	SurfaceSkin 0 2 "shieldsaw_HD.png"
	Scale 0.6333 0.6333 0.6333
	UseActorPitch
	UseActorRoll

	FrameIndex SFLY A 0 0
	FrameIndex SFLY B 0 1
	FrameIndex SFLY C 0 2
	FrameIndex SFLY D 0 3
	FrameIndex SFLY E 0 4
	FrameIndex SFLY F 0 5
	FrameIndex SFLY G 0 6
	FrameIndex SFLY H 0 7
	FrameIndex SFLY I 0 8
	FrameIndex SFLY J 0 9
	FrameIndex SFLY K 0 10
	FrameIndex SFLY L 0 11
	FrameIndex SFLY M 0 12
	FrameIndex SFLY N 0 13
	FrameIndex SFLY O 0 14
	FrameIndex SFLY P 0 15
}

	FrameIndex SLCK C 0 0
	AngleOffset 270.0
	FrameIndex SLCK D 0 0

// ---- stowed on your back -------------------------------------------------
// A WORLD model, not a psprite: it hangs off the body, and a psprite is drawn
// in a HAND's frame -- there is no hand at your shoulder blade. UseActorPitch
// and UseActorRoll let the mount lay it flat against your back at whatever
// angle the sliders end up wanting.
//
// Frame 0 of both meshes: shield closed, blade stowed. The right pose for
// something you are carrying rather than using.
//
// SCALE 0.34, THE SIZE OF THE SHIELD IN YOUR HAND (fixed 2026-09-14). The held
// and forearm blocks draw this mesh on the hand path, where one mesh unit is
// 0.34 map units: 57.94 across = 19.70. This block is a plain world actor, where
// one mesh unit is one map unit, so at 1.0 the shield on your back was 57.9
// across -- 2.9 times the one you carry. rs_ss_mount_scale still multiplies it
// (A_SetScale on the actor).
Model RS_ShieldStowActor
{
	Path "models/shieldsaw"
	Model 0 "shield.md3"
	Skin 0 "shieldsaw_b_HD.png"
	Model 1 "shieldsaw.md3"
	Skin 1 "shieldsaw_HD.png"
	Scale 0.34 0.34 0.34
	UseActorPitch
	UseActorRoll

	FrameIndex SSTW A 0 0	FrameIndex SSTW A 1 0
}

// ---- the mount marker ----------------------------------------------------
// HardPoints' wireframe diamond, so the shoulder mount reads the same way
// every other anchor in the family does. Scale is in map units and the mesh is
// radius 1, so this is the marker's real size.
//
// A SKIN, BECAUSE WITHOUT ONE IT DREW NOTHING (fixed 2026-09-14). The OBJ names
// no material and the block named no skin, and a model with no skin that
// resolves is skipped without a word. shieldsaw_glow.png, the shield's own glow,
// is the texture an Add-rendered marker wants. A second Path line so the skin is
// found beside the shield's textures.
Model RS_ShieldMountMarker
{
	Path "models"
	Model 0 "rs_ss_diamond.obj"
	Path "models/shieldsaw"
	Skin 0 "shieldsaw_glow.png"
	Scale 4.0 4.0 4.0
	FrameIndex SSMK A 0 0
}

// ---- THE WORLD PROPS --------------------------------------------------------
//
// The same shield and saw the psprite blocks above draw, on an actor that rides
// the controller instead. FollowMainHand / FollowOffHand put the mesh in that
// hand's frame and the renderer resolves it at DRAW rate -- which is why a prop
// on a moving hand is as smooth as the hand, where script placing it once a tic
// would visibly swim behind it.
//
// NO HAND MESH. The psprite blocks carry hand.md3 as model 2 because a player
// sprite has to draw its own hand or there is none. RS_WorldHands draws both
// hands for every weapon now, so including one here would put a second hand
// inside the first.
//
// The frame table is SSNH's, unchanged: model 0 is the shield opening 0->3,
// model 1 is the blade orbiting 0->7. The weapon's state machine still drives
// which letter, exactly as it did when it was driving a psprite.

Model RS_ShieldSawProp

	Path "models/shieldsaw"
	Model 0 "shield.md3"
	Skin 0 "shieldsaw_b_HD.png"
	Model 1 "shieldsaw.md3"
	Skin 1 "shieldsaw_HD.png"

	NOAUTOREVERSE
	FollowMainHand

	FrameIndex SSNH C 0 2	FrameIndex SSNH C 1 2
